C Calculator Documentation

A simple calculator written in C that runs in your terminal.

About

C Calculator is a lightweight command-line calculator written in the C programming language. It allows users to enter mathematical expressions directly into their terminal.

Installation

1. Create the source file

Open your terminal and put this in:

 main.c <<'EOF'

void spaces(char **p) {
    while (isspace((unsigned char)**p)) (*p)++;
}

double expression(char **p, int *error);

double number(char **p, int *error) {
    spaces(p);

    if (**p == '+') {
        (*p)++;
        return number(p, error);
    }

    if (**p == '-') {
        (*p)++;
        return -number(p, error);
    }

    if (**p == '(') {
        (*p)++;
        double result = expression(p, error);
        spaces(p);

        if (**p != ')') {
            *error = 1;
            return 0;
        }

        (*p)++;
        return result;
    }

    char *end;
    double value = strtod(*p, &end);

    if (end == *p) {
        *error = 1;
        return 0;
    }

    *p = end;
    return value;
}

double term(char **p, int *error) {
    double result = number(p, error);

    while (!*error) {
        spaces(p);

        char op = **p;
        if (op != '*' && op != '/') break;

        (*p)++;
        double next = number(p, error);

        if (*error) break;

        if (op == '*') {
            result *= next;
        } else {
            if (next == 0) {
                *error = 2;
                return 0;
            }
            result /= next;
        }
    }

    return result;
}

double expression(char **p, int *error) {
    double result = term(p, error);

    while (!*error) {
        spaces(p);

        char op = **p;
        if (op != '+' && op != '-') break;

        (*p)++;
        double next = term(p, error);

        if (*error) break;

        if (op == '+') result += next;
        else result -= next;
    }

    return result;
}

int main(void) {
    char input[512];

    printf("============================\n");
    printf("       C CALCULATOR\n");
    printf("============================\n");

    while (1) {
        printf("\nMath: ");
        fflush(stdout);

        if (!fgets(input, sizeof(input), stdin)) break;

        if (strcmp(input, "q\n") == 0 ||
            strcmp(input, "q") == 0) {
            break;
        }

        char *p = input;
        int error = 0;

        double result = expression(&p, &error);
        spaces(&p);

        if (error == 2) {
            printf("Cannot divide by zero.\n");
        } else if (error || *p != '\0') {
            printf("Invalid equation. Try again.\n");
        } else {
            printf("%.10g\n", result);
        }
    }

    printf("\nCalculator closed.\n");
    return 0;
}
EOF

2. Compile

Compile the program using Clang:

clang main.c -o calculator

3. Run

Start the calculator:

./calculator

Supported Operations

  • Addition (+)
  • Subtraction (-)
  • Multiplication (*)
  • Division (/)
  • Parentheses ( )
  • Decimal numbers
  • Negative numbers

Quitting

Type q and press Enter to close the calculator.

Supported Systems

  • Linux
  • Windows Terminal (with a C compiler)
  • macOS
  • Termux
  • ChromeOS Linux environment

Features

  • Written in C
  • 64-bit floating-point calculations
  • Multiple numbers in one expression
  • Basic mathematical operations
  • Error handling
  • Division-by-zero detection
  • Simple terminal interface
  • Quit using Q
